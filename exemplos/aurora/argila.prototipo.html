Quadro		Colaboradores ativos no último dia do período: admitidos até a data menos desligados até a data.inteiro
VAR _fim = MAX(dCalendario[Data])
VAR _entradas =
    CALCULATE(
        COUNTROWS(dColaborador),
        dColaborador[Data de Admissão] <= _fim,
        REMOVEFILTERS(dCalendario)
    )
VAR _saidas =
    CALCULATE(
        COUNTROWS(fDesligamento),
        fDesligamento[Data do Desligamento] <= _fim,
        REMOVEFILTERS(dCalendario)
    )
RETURN
    IF(MIN(dCalendario[Data]) > TODAY(), BLANK(), _entradas - _saidas)
